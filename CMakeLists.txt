cmake_minimum_required(VERSION 3.10)
project(Diary C)

# Build with:
#   cmake -B build -DCMAKE_TOOLCHAIN_FILE=<Retro68-build>/toolchain/m68k-apple-macos/cmake/retro68.toolchain.cmake
#   cmake --build build
# The finished app (with its custom icon) ends up in build/dist/.

add_application(Diary
    CREATOR "DIAR"
    diary.c
    net.c       # Diary Server client (MacTCP)
    diary.r
    icons.r     # generated by tools/make_icons.py
)

set_target_properties(Diary PROPERTIES COMPILE_OPTIONS -ffunction-sections)
set_target_properties(Diary PROPERTIES LINK_FLAGS "-Wl,-gc-sections -Wl,--mac-single")

# Retro68 can't set the Finder's hasBundle flag, which the Finder needs to
# show the custom icon. Set it in the MacBinary file, then put that copy
# into the disk image too.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(RETRO68_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
add_custom_command(
    OUTPUT dist/Diary.bin dist/Diary.dsk
    COMMAND ${CMAKE_COMMAND} -E make_directory dist
    COMMAND ${CMAKE_COMMAND} -E copy Diary.bin dist/Diary.bin
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/set_bundle_bit.py dist/Diary.bin
    COMMAND ${CMAKE_COMMAND} -E copy Diary.dsk dist/Diary.dsk
    COMMAND ${RETRO68_BIN}/hmount dist/Diary.dsk > /dev/null
    COMMAND ${RETRO68_BIN}/hdel :Diary
    COMMAND ${RETRO68_BIN}/hcopy -m dist/Diary.bin :Diary
    COMMAND ${RETRO68_BIN}/humount
    DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/Diary.bin ${CMAKE_SOURCE_DIR}/tools/set_bundle_bit.py
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    VERBATIM)
add_custom_target(Diary_dist ALL DEPENDS dist/Diary.bin dist/Diary.dsk)

# cmake --build build --target emulator
# puts the app into the folder Basilisk II shares with the Mac (ExtFS).
set(BASILISK_SHARED "$ENV{HOME}/BasiliskII/Shared" CACHE PATH
    "Folder shared with Basilisk II")
add_custom_target(emulator
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/macbinary_to_extfs.py
            dist/Diary.bin ${BASILISK_SHARED}
    DEPENDS Diary_dist
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    VERBATIM)
